# 2026-08-03

评论区通用表结构

```sql

CREATE TABLE `im_comment` (
  `comment_id` int NOT NULL AUTO_INCREMENT COMMENT '意见ID(主键)',
  `instance_guid` varchar(36)  COMMENT '业务单据主键',
  `fn_code` varchar(100) NULL COMMENT '业务单据FnCode',
  `comment_content` LONGtext NOT NULL COMMENT '意见内容',
  `reviewer_id` int NOT NULL COMMENT '评论人ID',
  `parent_id` bigint DEFAULT NULL COMMENT '父意见ID(支持回复功能)',
  `create_by` int NOT NULL COMMENT '创建人ID',
  `last_update_by` int NOT NULL COMMENT '最后更新人ID',
  `create_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `last_update_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '最后更新时间',
  `delete_by` int DEFAULT NULL COMMENT '删除人ID',
  `delete_flag` int NOT NULL DEFAULT '0' COMMENT '删除状态: 0-正常, -1-已删除',
  `delete_date` datetime DEFAULT NULL COMMENT '删除时间',
  `row_guid` varchar(36) NOT NULL COMMENT '唯一标识符',
  PRIMARY KEY (`comment_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='评论区评论内容';

CREATE TABLE `im_comment_assign` (
  `comment_guid` varchar(36)  COMMENT '明细主表guid',
  `instance_guid` varchar(36)  COMMENT '业务单据主键',
  `fn_code` varchar(100) NULL COMMENT '业务单据FnCode',
  `comment_content` text NOT NULL COMMENT '意见内容',
  `user_id` int NOT NULL COMMENT '抄送人',
  `create_by` int NOT NULL COMMENT '创建人ID',
  `last_update_by` int NOT NULL COMMENT '最后更新人ID',
  `create_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `last_update_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '最后更新时间',
  `delete_by` int DEFAULT NULL COMMENT '删除人ID',
  `delete_flag` int NOT NULL DEFAULT '0' COMMENT '删除状态: 0-正常, -1-已删除',
  `delete_date` datetime DEFAULT NULL COMMENT '删除时间',
  `row_guid` varchar(36) NOT NULL COMMENT '唯一标识符',
  PRIMARY KEY (`row_guid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='评论区抄送人';


CREATE TABLE `fs_imcomment` (
  `comment_guid` varchar(36)  COMMENT '主表Guid',
  `file_guid` varchar(36)  COMMENT '文件Guid',
  `fn_code` varchar(100) NULL COMMENT '业务单据FnCode',
  `comment_content` text NOT NULL COMMENT '意见内容',
  `reviewer_id` int NOT NULL COMMENT '评审人ID',
  `parent_id` bigint DEFAULT NULL COMMENT '父意见ID(支持回复功能)',
  `create_by` int NOT NULL COMMENT '创建人ID',
  `last_update_by` int NOT NULL COMMENT '最后更新人ID',
  `create_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `last_update_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '最后更新时间',
  `delete_by` int DEFAULT NULL COMMENT '删除人ID',
  `delete_flag` int NOT NULL DEFAULT '0' COMMENT '删除状态: 0-正常, -1-已删除',
  `delete_date` datetime DEFAULT NULL COMMENT '删除时间',
  `row_guid` varchar(36) NOT NULL COMMENT '唯一标识符',
  PRIMARY KEY (`row_guid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='评论区附件记录';

CREATE TABLE `fs_imcomment_content` (
  `file_guid` varchar(36)  COMMENT '文件Guid',
  `content`BLOB COMMENT '文件内容',
  `last_update_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '最后更新时间',
  `row_guid` varchar(36) NOT NULL COMMENT '唯一标识符',
  PRIMARY KEY (`row_guid`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='评论区附件内容';
```

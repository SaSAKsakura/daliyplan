评论区通用表结构

```sql

SET NAMES utf8mb4;
SET FOREIGN_KEY_CHECKS = 0;


DROP TABLE IF EXISTS `im_comment`;
CREATE TABLE `im_comment`  (
  `comment_id` int NOT NULL AUTO_INCREMENT COMMENT '意见ID(主键)',
  `instance_guid` varchar(36) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NULL DEFAULT NULL COMMENT '业务单据主键',
  `fn_code` varchar(100) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NULL DEFAULT NULL COMMENT '业务单据FnCode',
  `comment_content` longtext CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NULL COMMENT '意见内容',
  `reviewer_id` int NOT NULL COMMENT '评论人ID',
  `parent_id` bigint NULL DEFAULT NULL COMMENT '父意见ID(支持回复功能)',
  `create_by` int NOT NULL COMMENT '创建人ID',
  `last_update_by` int NOT NULL COMMENT '最后更新人ID',
  `create_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `last_update_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '最后更新时间',
  `delete_by` int NULL DEFAULT NULL COMMENT '删除人ID',
  `delete_flag` int NOT NULL DEFAULT 0 COMMENT '删除状态: 0-正常, -1-已删除',
  `delete_date` datetime NULL DEFAULT NULL COMMENT '删除时间',
  `row_guid` varchar(36) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL COMMENT '唯一标识符',
  `cancel_flag` int NULL DEFAULT 0 COMMENT '撤回标识', 
  PRIMARY KEY (`comment_id`) USING BTREE
) ENGINE = InnoDB AUTO_INCREMENT = 0 CHARACTER SET = utf8mb4 COLLATE = utf8mb4_0900_ai_ci COMMENT = '评论区评论内容' ROW_FORMAT = Dynamic;


DROP TABLE IF EXISTS `im_comment_assign`;
CREATE TABLE `im_comment_assign`  (
  `comment_guid` varchar(36) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NULL DEFAULT NULL COMMENT '明细主表guid',
  `instance_guid` varchar(36) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NULL DEFAULT NULL COMMENT '业务单据主键',
  `fn_code` varchar(100) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NULL DEFAULT NULL COMMENT '业务单据FnCode',
  `comment_content` text CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL COMMENT '意见内容',
  `user_id` int NOT NULL COMMENT '抄送人',
  `create_by` int NOT NULL COMMENT '创建人ID',
  `last_update_by` int NOT NULL COMMENT '最后更新人ID',
  `create_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `last_update_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '最后更新时间',
  `delete_by` int NULL DEFAULT NULL COMMENT '删除人ID',
  `delete_flag` int NOT NULL DEFAULT 0 COMMENT '删除状态: 0-正常, -1-已删除',
  `delete_date` datetime NULL DEFAULT NULL COMMENT '删除时间',
  `row_guid` varchar(36) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL COMMENT '唯一标识符',
  PRIMARY KEY (`row_guid`) USING BTREE
) ENGINE = InnoDB CHARACTER SET = utf8mb4 COLLATE = utf8mb4_0900_ai_ci COMMENT = '评论区抄送人' ROW_FORMAT = Dynamic;

SET FOREIGN_KEY_CHECKS = 1;

CREATE TABLE IF NOT EXISTS `fs_imcomment` (
    `row_guid` varchar(36) NOT NULL COMMENT '附件记录唯一标识符',

    `comment_guid` varchar(36) NOT NULL COMMENT '评论主表Guid',
    `file_guid` varchar(36) NOT NULL COMMENT '文件唯一Guid',
    `fn_code` varchar(100) DEFAULT NULL COMMENT '业务单据FnCode',

    `file_name` varchar(255) NOT NULL COMMENT '上传时的原始文件名',
    `file_ext` varchar(32) NOT NULL COMMENT '不带点的小写文件扩展名',
    `mime_type` varchar(255) DEFAULT NULL COMMENT 'HTTP MIME类型',
    `content_size` bigint NOT NULL DEFAULT 0 COMMENT '文件大小，单位：字节',
    `file_hash` varchar(128) DEFAULT NULL COMMENT '文件哈希值',

    `store_type` int NOT NULL COMMENT '存储方式：1-DB，2-FS，3-GRID_OS，4-MINIO，5-HCP，6-FK',
    `file_path` varchar(500) DEFAULT NULL COMMENT '文件系统相对路径，仅FS存储使用',

    `create_by` bigint NOT NULL COMMENT '创建人ID',
    `create_date` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `last_update_by` bigint NOT NULL COMMENT '最后更新人ID',
    `last_update_date` datetime NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP
        COMMENT '最后更新时间',

    `delete_flag` int NOT NULL DEFAULT 0 COMMENT '删除状态：0-正常，-1-已删除',
    `delete_by` bigint DEFAULT NULL COMMENT '删除人ID',
    `delete_date` datetime DEFAULT NULL COMMENT '删除时间',

    PRIMARY KEY (`row_guid`),

    UNIQUE KEY `uk_fs_imcomment_file_guid` (`file_guid`),

    KEY `idx_fs_imcomment_comment_delete`
        (`comment_guid`, `delete_flag`),

    KEY `idx_fs_imcomment_fn_code`
        (`fn_code`),

    KEY `idx_fs_imcomment_create_date`
        (`create_date`)
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_0900_ai_ci
  COMMENT='评论区附件元数据记录';


CREATE TABLE IF NOT EXISTS `fs_imcomment_content` (
    `row_guid` varchar(36) NOT NULL COMMENT '附件内容记录唯一标识符',
    `file_guid` varchar(36) NOT NULL COMMENT '对应附件文件Guid',

    `content` longblob DEFAULT NULL COMMENT '文件二进制内容，最大支持约4GB',

    `last_update_date` datetime NOT NULL
        DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP
        COMMENT '最后更新时间',

    PRIMARY KEY (`row_guid`)
) ENGINE=InnoDB
  DEFAULT CHARSET=utf8mb4
  COLLATE=utf8mb4_0900_ai_ci
  COMMENT='评论区附件数据库存储内容';		
		
```
